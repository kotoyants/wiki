Init repo
```bash linenums="1"
git init
```

Clone repo
```bash linenums="1"
# repo dir
git clone git@github.com:user/repo.git

# current dir
git clone git@github.com:user/repo.git .
```

Commands
```bash linenums="1"
git pull
git add .
git commit -m "Update"
git commit --amend --no-edit
git push
git push -f
git reset HEAD^
git reset --hard origin/main
git diff
git diff -- index.html
git stash
git rebase main
```
